   108→private val FA_DIGITS = charArrayOf('۰', '۱', '۲', '۳', '۴', '۵', '۶', '۷', '۸', '۹')
   109→
   110→private fun faDigits(s: String): String = buildString {
   111→    for (c in s) append(if (c.isDigit()) FA_DIGITS[c - '0'] else c)
   112→}
   113→
   114→private fun faNum(n: Long): String = faDigits(n.toString())
   115→private fun faNum(n: Int): String = faDigits(n.toString())
   116→
   117→private fun fmtSpeed(mbps: Double): String {
   118→    val s = if (mbps >= 100) String.format(Locale.US, "%.0f", mbps)
   119→    else String.format(Locale.US, "%.1f", mbps)
   120→    return faDigits(s)
   121→}

   173→@Composable
   174→fun MainScreen() {
   175→    val context = LocalContext.current
   176→    val scope = androidx.compose.runtime.rememberCoroutineScope()
   177→
   178→    val phase by VpnState.phase.collectAsState()
   179→    val coverMsg by VpnState.coverMsg.collectAsState()
   180→    val serverName by VpnState.serverName.collectAsState()
   181→    val serverCountry by VpnState.serverCountry.collectAsState()
   182→    val errorMessage by VpnState.errorMessage.collectAsState()
   183→    val rxBytes by VpnState.rxBytes.collectAsState()
   184→    val txBytes by VpnState.txBytes.collectAsState()
   185→
   186→    var showAdmin by remember { mutableStateOf(false) }
   187→    var configured by remember { mutableStateOf(AppPrefs.isConfigured(context)) }
   188→    var pendingServer by remember { mutableStateOf<ParsedServer?>(null) }
   189→
   190→    // connection timer
   191→    var seconds by remember { mutableIntStateOf(0) }
   192→    LaunchedEffect(phase) {
   193→        if (phase == Phase.CONNECTED) {
   194→            seconds = 0
   195→            while (true) {
   196→                kotlinx.coroutines.delay(1000)
   197→                seconds++
   198→            }
   199→        }
   200→    }
   201→
   202→    // live speeds from cumulative tunnel counters
   203→    var lastRx by remember { mutableLongStateOf(0L) }
   204→    var lastTx by remember { mutableLongStateOf(0L) }
   205→    var downMbps by remember { mutableStateOf(0.0) }
   206→    var upMbps by remember { mutableStateOf(0.0) }
   207→    LaunchedEffect(phase) {
   208→        if (phase == Phase.CONNECTED) {
   209→            lastRx = 0L; lastTx = 0L
   210→            while (true) {
   211→                kotlinx.coroutines.delay(1000)
   212→                val drx = rxBytes - lastRx
   213→                val dtx = txBytes - lastTx
   214→                if (lastRx > 0L || lastTx > 0L) {
   215→                    downMbps = drx.coerceAtLeast(0L) * 8.0 / 1_000_000.0
   216→                    upMbps = dtx.coerceAtLeast(0L) * 8.0 / 1_000_000.0
   217→                }
   218→                lastRx = rxBytes; lastTx = txBytes
   219→            }
   220→        }
   221→    }
   222→
   223→    // cover messages while busy
   224→    LaunchedEffect(phase) {
   225→        while (phase == Phase.SCANNING || phase == Phase.CONNECTING) {
   226→            VpnState.coverMsg.value =
   227→                (if (phase == Phase.SCANNING) SCAN_MSGS else CONNECT_MSGS).random()
   228→            kotlinx.coroutines.delay(1400)
   229→        }
   230→    }

   269→    fun startConnect() {
   270→        requestNotificationsIfNeeded()
   271→        scope.launch(kotlinx.coroutines.Dispatchers.Main) {
   272→            VpnState.reset()
   273→            VpnState.phase.value = Phase.SCANNING
   274→            try {
   275→                if (!AppPrefs.isConfigured(context)) throw SetupRequired()
   276→                val cfgs = kotlinx.coroutines.withContext(kotlinx.coroutines.Dispatchers.IO) {
   277→                    MianboorApi.fetchConfigs(
   278→                        AppPrefs.serverUrl(context),
   279→                        AppPrefs.clientToken(context)
   280→                    )
   281→                }
   282→                val parsed = cfgs.mapNotNull {
   283→                    XrayConfigBuilder.parseLink(it.link, it.name, it.country)
   284→                }
   285→                if (parsed.isEmpty()) throw IllegalStateException("هیچ کانفیگ فعالی موجود نیست")
   286→
   287→                val best = kotlinx.coroutines.withContext(kotlinx.coroutines.Dispatchers.IO) {
   288→                    ServerTester.testAll(parsed)
   289→                } ?: throw IllegalStateException("هیچ سروری در دسترس نیست؛ اینترنتت رو بررسی کن")
   290→
   291→                VpnState.serverName.value = best.name
   292→                VpnState.serverCountry.value = best.country
   293→
   294→                val prepare = VpnService.prepare(context)
   295→                if (prepare != null) {
   296→                    pendingServer = best
   297→                    vpnPermission.launch(prepare)
   298→                } else {
   299→                    launchVpn(best)
   300→                }
   301→            } catch (e: SetupRequired) {
   302→                VpnState.phase.value = Phase.IDLE
   303→                VpnState.errorMessage.value = "اول تنظیمات سرور میانبر رو وارد کن"
   304→                showAdmin = true
   305→            } catch (e: Exception) {
   306→                VpnState.phase.value = Phase.IDLE
   307→                VpnState.errorMessage.value = when (e) {
   308→                    is IllegalStateException -> e.message ?: "اتصال برقرار نشد"
   309→                    is java.net.UnknownHostException -> "سرور میانبر پیدا نشد؛ آدرس رو بررسی کن"
   310→                    is java.io.IOException -> "ارتباط با سرور ممکن نشد؛ اینترنتت رو بررسی کن"
   311→                    else -> "اتصال برقرار نشد؛ یک بار دیگه امتحان کن"
   312→                }
   313→            }
   314→        }
   315→    }

   538→@Composable
   539→private fun PowerButton(phase: Phase, onTap: () -> Unit) {
   540→    val busy = phase == Phase.SCANNING || phase == Phase.CONNECTING || phase == Phase.DISCONNECTING
   541→    val connected = phase == Phase.CONNECTED
   542→    val glowColor = if (connected || busy) Neon else TextDim.copy(alpha = 0.35f)
   543→
   544→    val pulse = rememberInfiniteTransition(label = "pulse")
   545→    val ringScale by pulse.animateFloat(
   546→        0.94f, 1.10f,
   547→        infiniteRepeatable(tween(1700, easing = FastOutSlowInEasing), RepeatMode.Restart),
   548→        label = "ringScale"
   549→    )
   550→    val ringAlpha by pulse.animateFloat(
   551→        0.5f, 0.0f,
   552→        infiniteRepeatable(tween(1700, easing = FastOutSlowInEasing), RepeatMode.Restart),
   553→        label = "ringAlpha"
   554→    )
   555→    val spin = rememberInfiniteTransition(label = "spin")
   556→    val angle by spin.animateFloat(
   557→        0f, 360f,
   558→        infiniteRepeatable(tween(2600, easing = LinearEasing), RepeatMode.Restart),
   559→        label = "angle"
   560→    )
   561→
   562→    Box(
   563→        modifier = Modifier
   564→            .padding(top = 10.dp)
   565→            .size(240.dp)
   566→            .pointerInput(Unit) { detectTapGestures(onTap = { onTap() }) },
   567→        contentAlignment = Alignment.Center
   568→    ) {
   569→        Canvas(Modifier.fillMaxSize()) {
   570→            val cx = size.width / 2f
   571→            val cy = size.height / 2f
   572→            val r = size.minDimension * 0.30f
   573→
   574→            // soft outer halos
   575→            if (connected || busy) {
   576→                for (i in 0..2) {
   577→                    val scale = ringScale + i * 0.09f
   578→                    val alpha = (ringAlpha - i * 0.13f).coerceIn(0f, 1f)
   579→                    if (alpha > 0.01f) {
   580→                        drawCircle(
   581→                            color = Neon.copy(alpha = alpha),
   582→                            radius = r * scale,
   583→                            center = Offset(cx, cy),
   584→                            style = Stroke(width = 3.dp.toPx())
   585→                        )
   586→                    }
   587→                }
   588→            } else {
   589→                drawCircle(
   590→                    color = glowColor.copy(alpha = 0.15f),
   591→                    radius = r * 1.28f,
   592→                    center = Offset(cx, cy),
   593→                    style = Stroke(width = 2.dp.toPx())
   594→                )
   595→            }
   596→
   597→            // rotating dashes while busy
   598→            if (busy) {
   599→                rotate(angle, Offset(cx, cy)) {
   600→                    val dash = 14f
   601→                    val gap = 12f
   602→                    var a = 0f
   603→                    while (a < 360f) {
   604→                        drawArc(
   605→                            color = Neon.copy(alpha = 0.65f),
   606→                            startAngle = a,
   607→                            sweepAngle = dash,
   608→                            useCenter = false,
   609→                            topLeft = Offset(cx - r * 1.42f, cy - r * 1.42f),
   610→                            size = androidx.compose.ui.geometry.Size(r * 2.84f, r * 2.84f),
   611→                            style = Stroke(width = 3.dp.toPx(), cap = StrokeCap.Round)
   612→                        )
   613→                        a += dash + gap
   614→                    }
   615→                }
   616→            }
   617→
   618→            // main disc
   619→            drawCircle(
   620→                brush = Brush.radialGradient(
   621→                    colors = listOf(
   622→                        if (connected) Color(0xFF17A87A) else Color(0xFF123229),
   623→                        Color(0xFF071310)
   624→                    ),
   625→                    center = Offset(cx, cy),
   626→                    radius = r
   627→                ),
   628→                radius = r,
   629→                center = Offset(cx, cy)
   630→            )
   631→            drawCircle(
   632→                color = if (connected) Neon else Neon.copy(alpha = 0.55f),
   633→                radius = r,
   634→                center = Offset(cx, cy),
   635→                style = Stroke(width = 4.dp.toPx())
   636→            )
   637→
   638→            // power glyph: circle arc with gap on top + vertical line
   639→            val gr = r * 0.52f
   640→            val sw = 5.dp.toPx()
   641→            drawArc(
   642→                color = TextMain,
   643→                startAngle = -55f,
   644→                sweepAngle = 290f,
   645→                useCenter = false,
   646→                topLeft = Offset(cx - gr, cy - gr),
   647→                size = androidx.compose.ui.geometry.Size(gr * 2f, gr * 2f),
   648→                style = Stroke(width = sw, cap = StrokeCap.Round)
   649→            )
   650→            drawLine(
   651→                color = TextMain,
   652→                start = Offset(cx, cy - gr - sw * 1.1f),
   653→                end = Offset(cx, cy - gr * 0.42f),
   654→                strokeWidth = sw,
   655→                cap = StrokeCap.Round
   656→            )
   657→        }
   658→    }
   659→}
